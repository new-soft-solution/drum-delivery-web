// This file is a REFERENCE, not a live route (the leading underscore in the
// folder name keeps Next.js from ever treating "_example-real-endpoint" as
// a real path segment). Copy this pattern into a real `route.ts` once your
// backend has an endpoint for something — e.g. once "clients" moves from
// the local mock store (src/lib/drum-tracer/store.ts) to a real backend
// endpoint, `src/app/api/clients/route.ts` would look like this:
//
//   import { proxyToBackend } from "@/lib/api/proxyRequest";
//
//   export async function GET(request: Request) {
//     const { search } = new URL(request.url);
//     return proxyToBackend(request, `/api/v1/clients/${search}`);
//   }
//
//   export async function POST(request: Request) {
//     return proxyToBackend(request, "/api/v1/clients/");
//   }
//
// `proxyToBackend` takes care of: reading the access token cookie,
// silently refreshing it if expired, forwarding the request body/method,
// and relaying the backend's JSON response + status code back to the
// client. Because the /api/clients URL and response shape the frontend
// talks to don't change, none of the CRUDTable/Form/Details components
// need to change when you make this switch — only this file does.
